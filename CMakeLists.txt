cmake_minimum_required(VERSION 3.24)
project(SparkGrid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
if(EMSCRIPTEN)
  set(PLATFORM Web CACHE STRING "" FORCE)
endif()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
set(USE_AUDIO ON CACHE BOOL "" FORCE)
set(SUPPORT_MODULE_RAUDIO ON CACHE BOOL "" FORCE)
if(MSVC)
  add_compile_options(/utf-8)
endif()
FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(raylib)
add_library(game_logic src/game.cpp)
target_include_directories(game_logic PUBLIC src)
if(EMSCRIPTEN)
  add_executable(SparkGrid src/main.cpp src/ui.cpp src/audio.cpp src/skins.cpp)
  set_target_properties(SparkGrid PROPERTIES SUFFIX ".html" OUTPUT_NAME "index")
  target_link_options(SparkGrid PRIVATE
    "--shell-file=${CMAKE_SOURCE_DIR}/web/shell.html"
    "--preload-file=${CMAKE_SOURCE_DIR}/data@/data"
    "-sALLOW_MEMORY_GROWTH=1" "-sINITIAL_MEMORY=33554432"
    "-sEXPORTED_RUNTIME_METHODS=ccall,FS" "-sGL_ENABLE_GET_PROC_ADDRESS=1")
else()
  add_executable(SparkGrid WIN32 src/main.cpp src/ui.cpp src/audio.cpp src/skins.cpp)
endif()
target_link_libraries(SparkGrid PRIVATE game_logic raylib)
if(MSVC)
  target_compile_options(game_logic PRIVATE /W4)
  target_link_options(SparkGrid PRIVATE /ENTRY:mainCRTStartup)
  target_link_options(SparkGrid PRIVATE $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
endif()
if(EMSCRIPTEN)
  add_custom_command(TARGET SparkGrid POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_SOURCE_DIR}/web/camera.js ${CMAKE_SOURCE_DIR}/web/pixel.js $<TARGET_FILE_DIR:SparkGrid>
    COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_SOURCE_DIR}/data/FONT-LICENSE.txt $<TARGET_FILE_DIR:SparkGrid>
    COMMAND ${CMAKE_COMMAND} -E touch $<TARGET_FILE_DIR:SparkGrid>/.nojekyll)
else()
  add_custom_command(TARGET SparkGrid POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/data $<TARGET_FILE_DIR:SparkGrid>/data)
  enable_testing()
  add_executable(logic_tests tests/logic_tests.cpp)
  target_link_libraries(logic_tests PRIVATE game_logic)
  add_test(NAME mechanics COMMAND logic_tests ${CMAKE_SOURCE_DIR}/data/stage01.txt)
endif()
